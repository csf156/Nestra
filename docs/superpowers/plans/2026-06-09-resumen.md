### Aesthetic direction (frontend-design skill)

**Editorial / data journalism** — números grandes como protagonistas, jerarquía tipográfica clara, color semántico con precisión quirúrgica.

- **Selector de mes:** Control `◀  Junio 2026  ▶` — mes en tipografía grande (`1.5rem bold`), arrows como botones circulares minimalistas. Transición fade en el label al cambiar.
- **KPI cards:** Número principal grande (`1.75rem bold`, `color-primary` para ingresos/balance+, `color-danger` para gastos/balance-). Badge delta debajo: `↑ +12%` (verde tintado) / `↓ -8%` (rojo tintado) / `= sin cambio` (neutro). Grid 3-col en tablet+, `display:flex; overflow-x:auto; gap` en móvil (scroll horizontal suave).
- **Progress bars:** Custom CSS (no `<progress>` nativo). `height:6px`, `border-radius:9999px`, track `--bg-light-secondary`, fill con `color-mix` según nivel.
- **Semáforo:** verde `#22c55e`, ámbar `#f59e0b`, rojo `var(--color-danger, #ef4444)` — siempre acompañado de texto ("Bien" / "Cerca" / "Excedido").
- **Transición de mes:** Al cambiar, el contenido hace fade-out/fade-in (`opacity 0.15s`).

### Lógica IIFE detallada

```js
(function () {
  'use strict';

  var MESES = ['Enero','Febrero','Marzo','Abril','Mayo','Junio',
               'Julio','Agosto','Septiembre','Octubre','Noviembre','Diciembre'];

  var _hoy   = new Date();
  var _mes   = _hoy.getMonth() + 1;
  var _anio  = _hoy.getFullYear();
  var _txExp = null;   // lazy cache para export

  function $(id) { return document.getElementById(id); }

  function escHtml(s) {
    return String(s == null ? '' : s)
      .replace(/&/g,'&amp;').replace(/</g,'&lt;')
      .replace(/>/g,'&gt;').replace(/"/g,'&quot;');
  }

  // --- Helpers mes ---
  function mesPrevio(mes, anio) {
    return mes === 1 ? { mes: 12, anio: anio - 1 } : { mes: mes - 1, anio: anio };
  }
  function primerDia(mes, anio) {
    var m = mes < 10 ? '0' + mes : String(mes);
    return anio + '-' + m + '-01';
  }
  function ultimoDia(mes, anio) {
    var ultimo = new Date(anio, mes, 0).getDate();
    var m = mes < 10 ? '0' + mes : String(mes);
    var d = ultimo < 10 ? '0' + ultimo : String(ultimo);
    return anio + '-' + m + '-' + d;
  }
  function esMesActual() {
    return _mes === _hoy.getMonth() + 1 && _anio === _hoy.getFullYear();
  }
  function actualizarNav() {
    $('resMesLabel').textContent = MESES[_mes - 1] + ' ' + _anio;
    var sig = $('resMesSiguiente');
    sig.disabled = esMesActual();
    sig.setAttribute('aria-disabled', esMesActual() ? 'true' : 'false');
  }

  // --- Delta badge ---
  // delta(actual, prev) → { pct: number|null, dir: 'up'|'down'|'flat' }
  function calcDelta(actual, prev) {
    if (prev === 0) return { pct: null, dir: 'flat' };
    var pct = ((actual - prev) / Math.abs(prev)) * 100;
    return { pct: pct, dir: pct > 0.5 ? 'up' : pct < -0.5 ? 'down' : 'flat' };
  }
  function renderDelta(delta, invertir) {
    // invertir=true: para Gastos, subir es malo (rojo), bajar es bueno (verde)
    if (delta.pct === null) return '<span class="res-delta res-delta--flat">sin dato previo</span>';
    if (delta.dir === 'flat') return '<span class="res-delta res-delta--flat">sin cambio</span>';
    var up = delta.dir === 'up';
    var positivo = invertir ? !up : up;
    var signo = up ? '↑' : '↓';
    var cls = positivo ? 'res-delta--pos' : 'res-delta--neg';
    return '<span class="res-delta ' + cls + '">' + signo + ' ' + Math.abs(delta.pct).toFixed(1) + '%</span>';
  }

  // --- KPI card ---
  function kpiCard(label, valor, delta, invertir, esBalance) {
    var negativo = esBalance && valor < 0;
    var valorCls = negativo ? 'res-kpi-valor--neg' : '';
    return '<div class="res-kpi-card">' +
      '<p class="res-kpi-label">' + escHtml(label) + '</p>' +
      '<p class="res-kpi-valor ' + valorCls + '">' + escHtml(formatMonto(valor)) + '</p>' +
      renderDelta(delta, invertir) +
    '</div>';
  }

  // --- Render Hogar ---
  function renderHogar(actual, prev) {
    var el = $('resHogarKpis');
    el.innerHTML =
      kpiCard('Ingresos',  actual.ingresos, calcDelta(actual.ingresos, prev.ingresos), false, false) +
      kpiCard('Gastos',    actual.gastos,   calcDelta(actual.gastos,   prev.gastos),   true,  false) +
      kpiCard('Balance',   actual.balance,  calcDelta(actual.balance,  prev.balance),  false, true);
    $('resHogarSection').style.display = 'block';
  }

  // --- Render Personal ---
  function renderPersonal(actual, prev) {
    var el = $('resPersonalKpis');
    el.innerHTML =
      kpiCard('Ingresos',       actual.ingresos,        calcDelta(actual.ingresos,        prev.ingresos),        false, false) +
      kpiCard('Gastos',         actual.gastos,          calcDelta(actual.gastos,           prev.gastos),          true,  false) +
      kpiCard('Aporte al hogar',actual.aporte_realizado,calcDelta(actual.aporte_realizado, prev.aporte_realizado),false, false) +
      kpiCard('Balance',        actual.balance,         calcDelta(actual.balance,          prev.balance),         false, true);
    $('resPersonalSection').style.display = 'block';
  }

  // --- Render Categorías ---
  function nivelPct(pct) {
    if (pct === null) return 'sin-limite';
    if (pct >= 100) return 'excedido';
    if (pct >= 70)  return 'cerca';
    return 'bien';
  }
  var nivelLabel = { 'bien': 'Bien', 'cerca': 'Cerca', 'excedido': 'Excedido', 'sin-limite': '' };
  var nivelOrder = { 'excedido': 0, 'cerca': 1, 'bien': 2, 'sin-limite': 3 };

  function renderCategorias(porCategoria, categoriasGasto, porCategoriaPrev) {
    // Enriquecer con limite_mensual
    var limMap = {};
    (categoriasGasto || []).forEach(function (c) { limMap[c.id] = c.limite_mensual || null; });

    // Mapa prev
    var prevMap = {};
    (porCategoriaPrev || []).forEach(function (c) { prevMap[c.categoria_id] = c.total; });

    var items = (porCategoria || []).map(function (c) {
      var lim = limMap[c.categoria_id];
      var pct = (lim && lim > 0) ? (c.total / lim) * 100 : null;
      return {
        nombre:   c.nombre,
        total:    c.total,
        limite:   lim,
        pct:      pct,
        nivel:    nivelPct(pct),
        totalPrev: prevMap[c.categoria_id] || 0,
      };
    }).sort(function (a, b) {
      var od = nivelOrder[a.nivel] - nivelOrder[b.nivel];
      return od !== 0 ? od : b.total - a.total;
    });

    if (!items.length) {
      $('resCatVacio').style.display = 'block';
      $('resCatSection').style.display = 'block';
      return;
    }
    $('resCatVacio').style.display = 'none';

    // Mobile cards
    $('resCatLista').innerHTML = items.map(function (it) {
      var pctNum = it.pct !== null ? Math.min(it.pct, 100) : 0;
      var delta  = calcDelta(it.total, it.totalPrev);
      return '<div class="res-cat-card res-cat-card--' + it.nivel + '">' +
        '<div class="res-cat-card-header">' +
          '<span class="res-cat-nombre">' + escHtml(it.nombre) + '</span>' +
          '<span class="res-cat-monto">' + escHtml(formatMonto(it.total)) + '</span>' +
        '</div>' +
        (it.limite !== null ?
          '<div class="res-cat-bar-track"><div class="res-cat-bar-fill res-cat-bar--' + it.nivel + '" style="width:' + pctNum.toFixed(1) + '%"></div></div>' +
          '<div class="res-cat-card-footer">' +
            '<span class="res-cat-badge res-cat-badge--' + it.nivel + '">' + nivelLabel[it.nivel] + ' ' + it.pct.toFixed(0) + '%</span>' +
            '<span class="res-cat-limite">límite ' + escHtml(formatMonto(it.limite)) + '</span>' +
          '</div>' :
          '<div class="res-cat-card-footer">' +
            renderDelta(delta, true) +
          '</div>'
        ) +
      '</div>';
    }).join('');

    // Desktop tabla
    $('resCatTablaWrap').innerHTML =
      '<table class="res-tabla" aria-label="Gastos por categoría">' +
      '<thead><tr>' +
        '<th scope="col">Categoría</th>' +
        '<th scope="col" class="res-th-num">Gasto</th>' +
        '<th scope="col" class="res-th-num">Límite</th>' +
        '<th scope="col">% del límite</th>' +
        '<th scope="col" class="res-th-num">vs. anterior</th>' +
      '</tr></thead>' +
      '<tbody>' +
      items.map(function (it) {
        var pctNum = it.pct !== null ? Math.min(it.pct, 100) : 0;
        var delta  = calcDelta(it.total, it.totalPrev);
        return '<tr>' +
          '<td>' + escHtml(it.nombre) + '</td>' +
          '<td class="res-td-num">' + escHtml(formatMonto(it.total)) + '</td>' +
          '<td class="res-td-num">' + (it.limite !== null ? escHtml(formatMonto(it.limite)) : '—') + '</td>' +
          '<td>' +
            (it.limite !== null ?
              '<div class="res-tabla-bar-wrap">' +
                '<div class="res-cat-bar-track res-cat-bar-track--tabla"><div class="res-cat-bar-fill res-cat-bar--' + it.nivel + '" style="width:' + pctNum.toFixed(1) + '%"></div></div>' +
                '<span class="res-cat-badge res-cat-badge--' + it.nivel + '">' + nivelLabel[it.nivel] + ' ' + it.pct.toFixed(0) + '%</span>' +
              '</div>' : '—') +
          '</td>' +
          '<td class="res-td-num">' + renderDelta(delta, true) + '</td>' +
        '</tr>';
      }).join('') +
      '</tbody></table>';

    $('resCatSection').style.display = 'block';
  }

  // --- Render Metas ---
  function renderMetas(metas, mes, anio) {
    var desde = primerDia(mes, anio);
    var hasta = ultimoDia(mes, anio);
    var hoyIso = _hoy.toISOString().split('T')[0];

    var completadas = (metas || []).filter(function (m) {
      return m.estado === 'completada' && m.fecha_limite >= desde && m.fecha_limite <= hasta;
    });
    var vencidas = (metas || []).filter(function (m) {
      return m.estado !== 'completada' && m.fecha_limite && m.fecha_limite <= hasta && m.fecha_limite <= hoyIso;
    });

    var ambosVacios = !completadas.length && !vencidas.length;
    $('resMetasVacio').style.display = ambosVacios ? 'block' : 'none';

    function renderGrupo(lista, listaId, vacioId, labelId, toggleId) {
      var label = $(labelId);
      if (!label) return;
      var ul = $(listaId);
      var vaciop = $(vacioId);
      var toggle = $(toggleId);
      var nombre = toggle.querySelector('span');
      nombre.textContent = nombre.textContent.replace(/\(\d+\)/, '(' + lista.length + ')');

      if (!lista.length) {
        ul.innerHTML = '';
        vaciop.style.display = 'block';
        return;
      }
      vaciop.style.display = 'none';
      ul.innerHTML = lista.map(function (m) {
        var progPct = m.monto_objetivo > 0 ? Math.min((m.monto_actual / m.monto_objetivo) * 100, 100) : 0;
        return '<li class="res-meta-item">' +
          '<div class="res-meta-header">' +
            '<span class="res-meta-nombre">' + escHtml(m.nombre) + '</span>' +
            (m.fecha_limite ? '<span class="res-meta-fecha">' + escHtml(formatFecha(m.fecha_limite)) + '</span>' : '') +
          '</div>' +
          '<div class="res-cat-bar-track">' +
            '<div class="res-cat-bar-fill res-cat-bar--bien" style="width:' + progPct.toFixed(1) + '%"></div>' +
          '</div>' +
          '<div class="res-meta-footer">' +
            '<span class="res-meta-prog">' + escHtml(formatMonto(m.monto_actual)) + ' / ' + escHtml(formatMonto(m.monto_objetivo)) + '</span>' +
            '<span class="res-meta-pct">' + progPct.toFixed(0) + '%</span>' +
          '</div>' +
        '</li>';
      }).join('');
    }

    renderGrupo(completadas, 'resMetasCompLista', 'resMetasCompVacio', 'resMetasCompLabel', 'resMetasCompToggle');
    renderGrupo(vencidas,    'resMetasVencLista',  'resMetasVencVacio',  'resMetasVencLabel',  'resMetasVencToggle');

    $('resMetasSection').style.display = 'block';
  }

  // --- Toast ---
  var _toastTimer = null;
  function mostrarToast(msg, ms) {
    clearTimeout(_toastTimer);
    $('restoastMsg').textContent = msg;
    $('restoast').style.display = 'flex';
    _toastTimer = setTimeout(function () { $('restoast').style.display = 'none'; }, ms || 4000);
  }

  // --- Fade transition ---
  function fadeContent(fn) {
    var secs = ['resHogarSection','resPersonalSection','resCatSection','resMetasSection','resExport'];
    secs.forEach(function (id) {
      var el = $(id);
      if (el) { el.style.opacity = '0'; el.style.transition = 'opacity 0.15s'; }
    });
    setTimeout(function () {
      fn();
      secs.forEach(function (id) {
        var el = $(id);
        if (el && el.style.display !== 'none') { el.style.opacity = '1'; }
      });
    }, 150);
  }

  // --- Cargar ---
  async function cargar() {
    $('resLoading').style.display = 'flex';
    ['resHogarSection','resPersonalSection','resCatSection','resMetasSection','resExport'].forEach(function (id) {
      var el = $(id); if (el) { el.style.display = 'none'; el.style.opacity = '0'; }
    });
    _txExp = null;

    try {
      var prev = mesPrevio(_mes, _anio);
      var res = await Promise.all([
        getResumenMensual(_mes, _anio),
        getResumenMensual(prev.mes, prev.anio),
        getCategorias('gasto'),
        getMetas(),
      ]);
      var actual = res[0];
      var anterior = res[1];
      var cats = res[2];
      var metas = res[3];

      fadeContent(function () {
        renderHogar(actual.hogar, anterior.hogar);
        renderPersonal(actual.personal, anterior.personal);
        renderCategorias(actual.porCategoria, cats, anterior.porCategoria);
        renderMetas(metas, _mes, _anio);
        $('resExport').style.display = 'flex';
      });
    } catch (err) {
      console.error('resumen cargar:', err);
      mostrarToast('No se pudieron cargar los datos.', 5000);
    } finally {
      $('resLoading').style.display = 'none';
    }
  }

  // --- Navegación mes ---
  $('resMesAnterior').addEventListener('click', function () {
    var p = mesPrevio(_mes, _anio);
    _mes = p.mes; _anio = p.anio;
    actualizarNav();
    cargar();
  });
  $('resMesSiguiente').addEventListener('click', function () {
    if (esMesActual()) return;
    _mes++; if (_mes > 12) { _mes = 1; _anio++; }
    actualizarNav();
    cargar();
  });

  // --- Toggles metas ---
  ['resMetasCompToggle','resMetasVencToggle'].forEach(function (toggleId) {
    var panel = toggleId === 'resMetasCompToggle' ? 'resMetasCompPanel' : 'resMetasVencPanel';
    $(toggleId).addEventListener('click', function () {
      var expanded = $(toggleId).getAttribute('aria-expanded') === 'true';
      $(toggleId).setAttribute('aria-expanded', expanded ? 'false' : 'true');
      $(panel).style.display = expanded ? 'none' : 'block';
      $(toggleId).querySelector('.res-toggle-caret').textContent = expanded ? '⌄' : '⌃';
    });
  });

  // --- Export ---
  $('resBtnPdf').addEventListener('click', function () {
    exportador.exportPDF();
  });

  $('resBtnXlsx').addEventListener('click', async function () {
    var btn = $('resBtnXlsx');
    btn.disabled = true;
    mostrarToast('Exportando…', 3000);
    try {
      if (!_txExp) {
        _txExp = await getTransacciones({ fecha_desde: primerDia(_mes, _anio), fecha_hasta: ultimoDia(_mes, _anio) });
      }
      var res = exportador.exportXLSX(_txExp);
      if (res.ok) {
        mostrarToast('Descargado (' + res.count + ' filas)', 4000);
      } else if (res.reason === 'sin-datos') {
        mostrarToast('Sin transacciones en ' + MESES[_mes - 1] + ' ' + _anio, 4000);
      } else {
        mostrarToast('Error al exportar: ' + (res.reason || 'desconocido'), 5000);
      }
    } catch (err) {
      mostrarToast('No se pudo exportar.', 5000);
    } finally {
      btn.disabled = false;
    }
  });

  // --- Init ---
  actualizarNav();
  cargar();

})();
```
